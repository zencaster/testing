Agar setiap transaksi otomatis masuk ke baris paling bawah pada Google Sheet Anda, tempelkan URL Web App Google Apps Script di bawah ini:

Buka Sheet
Kode Apps Script (Salin ke Extensions > Apps Script) 
const SHEET_ID = '18hYx94mb6zrObubgV4nZ8D-u83mYPNXXNBBfaPyAbxI';

function getTargetSheet() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheet = ss.getSheets()[0];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Tanggal','Type','Nominal','Keterangan','Kategori','User','Timestamp']);
  }
  return sheet;
}

function doGet() {
  const sheet = getTargetSheet();
  const data = sheet.getDataRange().getValues();
  const rows = [];
  for (let i = 1; i < data.length; i++) {
    if (!data[i][0] && !data[i][2]) continue;
    let d = data[i][0];
    if (d instanceof Date) d = Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    rows.push({
      tanggal: String(d),
      type: String(data[i][1] || 'Pengeluaran'),
      nominal: Number(data[i][2]) || 0,
      keterangan: String(data[i][3] || ''),
      kategori: String(data[i][4] || 'OTHER'),
      user: String(data[i][5] || '')
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: rows })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = getTargetSheet();
  const p = JSON.parse(e.postData.contents);
  const ts = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  sheet.appendRow([p.tanggal, p.type, Number(p.nominal), p.keterangan, p.kategori, p.user, ts]);
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
}
